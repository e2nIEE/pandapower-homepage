{%- comment -%} Recursive menu renderer. Params: items (list). Uses lang/prefix from context. Marks the current page with aria-current="page" (exact match, or section match for non-root urls, e.g. /news/ stays active on /news/2026/...). Ancestors of the current page are highlighted via CSS (:has). {%- endcomment -%}
    {%- for item in include.items -%} {%- assign is_current = false -%} {%- unless item.external or item.url contains "#" -%} {%- assign full = prefix | append: item.url -%} {%- assign n = full | size -%} {%- assign head = page.url | slice: 0, n -%} {%- if page.url == full -%} {%- assign is_current = true -%} {%- elsif item.url != "/" and head == full -%} {%- assign is_current = true -%} {%- endif -%} {%- endunless -%} {%- if item.url -%} {{ item.title[lang] }} {%- else -%} {{ item.title[lang] }} {%- endif -%} {%- if item.children -%}{% include nav-items.html items=item.children %}{%- endif -%} {%- endfor -%}
